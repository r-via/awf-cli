func longestInputKey(inputs map[string]any) string {
	keys := make([]string, 0, len(inputs))
	for k := range inputs {
		keys = append(keys, k)
	}

	sort.Slice(keys, func(i, j int) bool {
		li := valueLength(inputs[keys[i]])
		lj := valueLength(inputs[keys[j]])
		if li != lj {
			return li > lj
		}
		return keys[i] < keys[j]
	})

	return keys[0]
}
