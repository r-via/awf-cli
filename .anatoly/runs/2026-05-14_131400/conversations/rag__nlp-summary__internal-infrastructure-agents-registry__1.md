func (r *AgentRegistry) Register(provider ports.AgentProvider) error {
	name := provider.Name()
	r.mu.Lock()
	defer r.mu.Unlock()

	if _, exists := r.providers[name]; exists {
		return fmt.Errorf("provider %q already registered", name)
	}

	r.providers[name] = provider
	return nil
}
