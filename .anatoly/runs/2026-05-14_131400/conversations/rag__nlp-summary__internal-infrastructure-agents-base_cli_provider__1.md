	output := make([]byte, 0, len(stdoutBytes)+len(stderrBytes))
	output = append(output, stdoutBytes...)

	return string(output)

	completedAt := time.Now()
	if filter != nil {

	}

		return nil, "", fmt.Errorf("%s execution failed: %w", b.name, err)
